    defer engine.deinit();

    , "native-semver785-retain-before-gc.mjs");

// Prepared native allocation-identity controls. These use the actual private
// Engine tag; no guest name/message match and no allocator-sweep pass claim.
fn throwSemverAllocation(context: ?*c.JSContext, _: c.JSValue, _: c_int, _: [*c]c.JSValue) callconv(.c) c.JSValue {
    const engine = em.Engine.fromContext(context.?);
    _ = engine.throwNativeOutOfMemory();
    const original = c.JS_GetException(context);
    const global = c.JS_GetGlobalObject(context);
    defer engine.freeValue(global);
    if (c.JS_DefinePropertyValueStr(context, global, "ownedSemverAllocation", c.JS_DupValue(context, original), c.JS_PROP_C_W_E) < 0) {
        engine.freeValue(original);
        return c.JS_Throw(context, c.JS_GetException(context));
    }
    return c.JS_Throw(context, original);
}
test "native semver785 forgiving catches retain exact fresh allocation reason" {
    for ([_][]const u8{ "valid", "validRange" }) |method| {
        const engine = try em.Engine.init(std.testing.allocator, .{ .host_await_timeout_ms = 10000 });
        defer engine.deinit();
        const global = c.JS_GetGlobalObject(engine.context);
        defer engine.freeValue(global);
        const js = @import("extensions/native_js_values.zig");
        try js.define(engine, global, "semver785API", try @import("extensions/native_semver785.zig").create(engine));
        try js.define(engine, global, "raiseOwnedSemverAllocation", try engine.checked(c.JS_NewCFunction(engine.context, throwSemverAllocation, "raiseOwnedSemverAllocation", 0)));
        try js.define(engine, global, "ownedSemverMethod", try sdk.text(engine, method));
        try std.testing.expectError(error.JavaScriptException, engine.evalModule(
            \\globalThis.ownedGuestCaught=false;const original=String.prototype.trim;String.prototype.trim=raiseOwnedSemverAllocation;try{try{semver785API[ownedSemverMethod]('1.2.3')}catch{globalThis.ownedGuestCaught=true}}finally{String.prototype.trim=original}
        , "native-semver785-fresh-allocation-identity.mjs"));
        const retained = try js.get(engine, global, "ownedSemverAllocation");
        defer engine.freeValue(retained);
        const actual = engine.captured_exception orelse return error.MissingSemverAllocationReason;
        try std.testing.expect(c.JS_IsStrictEqual(engine.context, retained, actual));
        try std.testing.expect(c.JS_IsUncatchableError(actual));
        const caught = try js.get(engine, global, "ownedGuestCaught");
        defer engine.freeValue(caught);
        try std.testing.expectEqual(@as(c_int, 0), c.JS_ToBool(engine.context, caught));
    }
