function live(index, action) {
  const log = [], sentinel = { tag: 'read-range-sentinel' };
  let value, failure;
  try { value = action(log, sentinel); }
  catch (error) { failure = error === sentinel ? { same: true } : { name: error.name, message: error.message }; }
  rows.push({ kind: 'live', index, ...(failure ? { failure } : { value }), log: [...log] });
}
function patch(target, key, descriptor, action) {
  const old = Object.getOwnPropertyDescriptor(target, key);
  Object.defineProperty(target, key, descriptor);
  try { return action(); } finally { Object.defineProperty(target, key, old); }
}
