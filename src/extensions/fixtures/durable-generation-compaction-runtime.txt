import{Harness,MemoryStorage,createRegistry,UserEntry,AssistantEntry,LiveDoc,UsageDoc}from'@earendil-works/pi-durable';const rows=[];
for(const scenario of ['blocking','background','manual-reset']){let requests=0,summaries=0;const reports=[],usage={input:1,output:2,cacheRead:0,cacheWrite:0,totalTokens:3,cost:{input:0,output:0,cacheRead:0,cacheWrite:0,total:0}},message=text=>({role:'assistant',provider:'p',model:'m',content:[{type:'text',text}],stopReason:'stop',usage});const models={getModel:()=>({id:'m',maxTokens:100,contextWindow:scenario==='blocking'?1000:2000}),streamSimple:()=>{requests++;return{async*[Symbol.asyncIterator](){yield{type:'start',partial:{content:[]}}},result:async()=>message('answer')}},completeSimple:async(model,context,options)=>{summaries++;if(options.cacheRetention!=='none'||!(options.signal instanceof AbortSignal))throw Error('summary request');return message('summary')}};const harness=await Harness.open(new MemoryStorage(),{registry:createRegistry(),models,settings:{compaction:{enabled:scenario!=='manual-reset',reserveTokens:100,backgroundTokens:1500,keepRecentTokens:0}},onReport:error=>reports.push(String(error))},{});try{const root=await harness.root({},{agent:{model:{provider:'p',modelId:'m'}}});await root.commit(async tx=>{for(let n=0;n<3;n++){await tx.appendEntry(UserEntry,root.id,{model:[{role:'user',content:'old'.repeat(400),timestamp:n}]});await tx.appendEntry(AssistantEntry,root.id,{model:[{...message('old answer'),usage:{...usage,totalTokens:800},timestamp:n}]})}},{});if(scenario==='manual-reset'){const id=await root.compact('retain important context',{}),task=await harness.waitForTask(id,{});if(task.state?.outcome?.status!=='completed')throw Error(JSON.stringify(task));await root.reset('start here',{});const view=await root.context({});if(view.messages.length!==1||view.messages[0].content!=='start here')throw Error(JSON.stringify(view));}else{const submission=await root.submit({type:'input',content:'new harness.snapshot(LiveDoc,root.id,{});if(!state.compactions?.length)break;await new Promise(resolve=>setTimeout(resolve,5))}const live=await harness.snapshot(LiveDoc,root.id,{}),ledger=await harness.snapshot(UsageDoc,root.id,{});if(summaries!==1||reports.length||Object.keys(live).length)throw Error(JSON.stringify({scenario,requests,summaries,reports,live}));rows.push({scenario,requests,summaries,tokens:ledger.models['p/m'].totalTokens,live});}finally{await harness.close({})}}
globalThis.generationCompactionRows=rows;
