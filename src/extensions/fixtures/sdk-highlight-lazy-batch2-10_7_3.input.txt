function graph(root){
  const seen=new Map();let next=0;
  function encode(value){
    if(value===undefined)return{$undefined:true};
    if(value===null||typeof value==='string'||typeof value==='boolean')return value;
    if(typeof value==='number')return Number.isFinite(value)?value:{$number:String(value)};
    if(typeof value==='function')return{$function:{name:value.name,length:value.length,keys:Reflect.ownKeys(value).map(String),prototype:Object.hasOwn(value,'prototype')}};
    if(typeof value!=='object')return{$other:typeof value,value:String(value)};
    if(seen.has(value))return{$ref:seen.get(value)};
    const id=next++;seen.set(value,id);
    if(value instanceof RegExp)return{$id:id,$regexp:value.source,flags:value.flags,lastIndex:value.lastIndex,frozen:Object.isFrozen(value)};
    return{$id:id,kind:Array.isArray(value)?'array':'object',frozen:Object.isFrozen(value),properties:Reflect.ownKeys(value).map(key=>{
      const d=Object.getOwnPropertyDescriptor(value,key);
      return{key:String(key),enumerable:d.enumerable,configurable:d.configurable,...('value'in d?{writable:d.writable,value:encode(d.value)}:{get:encode(d.get),set:encode(d.set)})};
    })};
  }
  return encode(root);
}
